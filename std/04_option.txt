union Option
    %some ?
    %none

fn mk_some Option v:?
    %some ~ $v
%none none

fn is_some Bool o:Option
    const always_true $true
    just just_false $false
    % ~ $o some;$always_true none;$just_false
fn is_none Bool o:Option
    chain2_1 ~ $o $is_some $not

fn map_opt Option f:<?,?> o:Option
    just just_none $none
    compose2_1 f_some $f $mk_some
    % ~ $o some;$f_some none;$just_none

fn unwrap_or ? when_none:? o:Option
    just just_when_none $when_none
    % ~ $o some;$id none;$just_when_none

fn map_or ? when_none:? f:<?,?> o:Option
    map_opt mapped $f $o
    unwrap_or ~ $when_none $mapped

fn lazy_map_or ? when_none:<?> f:<<?>,?> o:Option
    map_or res_f $when_none $f $o
    res_f ~

fn lazy_unwrap_or ? when_none:<?> o:Option
    lazy_map_or ~ $when_none $just $o

fn unwrap! ? o:Option
    lazy_unwrap_or ~ $error! $o

fn join_opt Option o:Option
    map_or ~ $none $id $o

fn bind_opt Option f:<Option,?> o:Option
    map_opt mapped $f $o
    join_opt ~ $mapped

union Affine
    %affine <Option,?>

fn mut_of? Option s:? a:Affine
    % a_f $a affine;$id
    a_f ~ $s

fn view? Option s:? a:Affine
    mut_of? maybe_m $s $a
    map_opt ~ $view_mut $maybe_m

fn view?! ? s:? a:Affine
    mut_of? maybe_m $s $a
    unwrap! ~ $maybe_m

fn set?? Option s:? a:Affine new_val:?
    apply2b set_new_val $set_mut $new_val
    mut_of? maybe_m $s $a
    map_opt ~ $set_new_val $maybe_m

fn set? ? s:? a:Affine new_val:?
    set?? maybe_res $s $a $new_val
    unwrap_or ~ $s $maybe_res

fn set?! ? s:? a:Affine new_val:?
    set?? maybe_res $s $a $new_val
    unwrap! ~ $maybe_res

fn over?? Option s:? a:Affine f:<?,?>
    apply2b over_f $over_mut $f
    mut_of? maybe_m $s $a
    map_opt ~ $over_f $maybe_m

fn over? ? s:? a:Affine f:<?,?>
    over?? maybe_res $s $a $f
    unwrap_or ~ $s $maybe_res

fn over?! ? s:? a:Affine f:<?,?>
    over?? maybe_res $s $a $f
    unwrap! ~ $maybe_res

fn lens_to_affine Affine lens:Lens
    % lens_f $lens lens;$id
    compose2_1 affine_f $lens_f $mk_some
    %affine ~ $affine_f
