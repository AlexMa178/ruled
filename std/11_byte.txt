fn byte_to_nat Nat b:#
    0# bit0 $b $true $false
    1# bit1 $b $true $false
    2# bit2 $b $true $false
    3# bit3 $b $true $false
    4# bit4 $b $true $false
    5# bit5 $b $true $false
    6# bit6 $b $true $false
    7# bit7 $b $true $false
    list_8 bits $bit0 $bit1 $bit2 $bit3 $bit4 $bit5 $bit6 $bit7
    trimr_while trim_bits $not $bits
    %nat_bits ~ $trim_bits
