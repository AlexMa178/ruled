    > $c.code
    id ~ #unit

fn read List
    fn read_char List
        < code B7
        Char char code;$code
        b7_to_nat code_n $code
        is_zero is_null $code_n
        just when_null $empty_list
        fn when_non_null List
            read_char rest
            push_front ~ $rest $char
        lazy_if ~ $is_null $when_null $when_non_null
    ^
    read_char ~

fn is_lower Bool c:Char
    b7_to_nat n $c.code
    b7_to_nat a_n $'a.code
    b7_to_nat z_n $'z.code
    ge_nat in_min $n $a_n
    le_nat in_max $n $z_n
    and ~ $in_min $in_max

fn is_upper Bool c:Char
    b7_to_nat n $c.code
    b7_to_nat a_n $'A.code
    b7_to_nat z_n $'Z.code
    ge_nat in_min $n $a_n
    le_nat in_max $n $z_n
    and ~ $in_min $in_max

fn to_upper Char c:Char
    is_lower in_bounds $c
    fn if_in Char
        b7_to_nat n $c.code
        set! new_bits $n.bits $5 #false
        Nat new_n bits;$new_bits
        nat_to_b7! new_code $new_n
        Char ~ code;$new_code
    just if_not_in $c
    lazy_if ~ $in_bounds $if_in $if_not_in

fn to_lower Char c:Char
    is_upper in_bounds $c
    fn if_in Char
        b7_to_nat n $c.code
        set! new_bits $n.bits $5 #true
        Nat new_n bits;$new_bits
        nat_to_b7! new_code $new_n
        Char ~ code;$new_code
    just if_not_in $c
    lazy_if ~ $in_bounds $if_in $if_not_in
