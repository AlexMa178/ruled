fn is_lower Bool c:Char
    byte_to_nat n $c.code
    byte_to_nat a_n $'a.code
    byte_to_nat z_n $'z.code
    ge_nat in_min $n $a_n
    le_nat in_max $n $z_n
    and ~ $in_min $in_max

fn is_upper Bool c:Char
    byte_to_nat n $c.code
    byte_to_nat a_n $'A.code
    byte_to_nat z_n $'Z.code
    ge_nat in_min $n $a_n
    le_nat in_max $n $z_n
    and ~ $in_min $in_max

fn to_upper Char c:Char
    is_lower in_bounds $c
    fn if_in Char
        byte_to_nat n $c.code
        set! new_bits $n.bits $5 $false
        Nat new_n bits;$new_bits
        nat_to_byte new_code $new_n
        Char ~ code;$new_code
    just if_not_in $c
    lazy_if ~ $in_bounds $if_in $if_not_in

fn to_lower Char c:Char
    is_upper in_bounds $c
    fn if_in Char
        byte_to_nat n $c.code
        set! new_bits $n.bits $5 $true
        Nat new_n bits;$new_bits
        nat_to_byte new_code $new_n

    lazy_if ~ $in_bounds $if_in $if_not_in
