union Pair
    %pair ? ?

fn fst ? p:Pair
    % ~ $p pair;$keep2a

fn snd ? p:Pair
    % ~ $p pair;$keep2b

fn map_both Pair p:Pair f_fst:<?,?> f_snd:<?,?>
    fn act Pair fst:? snd:?
        f_fst new_fst $fst
        f_snd new_snd $snd
        %pair ~ $new_fst $new_snd
    % ~ $p pair;$act

fn map_fst Pair p:Pair f:<?,?>
    map_both ~ $p $f $id

fn map_snd Pair p:Pair f:<?,?>
    map_both ~ $p $id $f

fn set_fst Pair p:Pair v:?
    const always_v $v
    map_fst ~ $p $always_v

fn set_snd Pair p:Pair v:?
    const always_v $v
    map_snd ~ $p $always_v
